cmake_minimum_required(VERSION 3.16)
project(RRTstarDFPFF VERSION 0.2.0 LANGUAGES CXX)
option(RRTSTARDFPFF_ENABLE_AVX2 "Build runtime-dispatched AVX2 kernels" ON)
set(RRTPFF_EIGEN_SOURCE "" CACHE PATH "Optional Eigen source tree for an offline build")
set(RRTPFF_YAML_SOURCE "" CACHE PATH "Optional yaml-cpp source tree for an offline build")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# The speed certificate relies on finite/NaN checks and IEEE arithmetic.
if("${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE}" MATCHES "-ffast-math|-Ofast|/fp:fast")
    message(FATAL_ERROR "Production interval certificates require strict floating-point flags")
endif()
if(RRTPFF_EIGEN_SOURCE)
    add_library(rrtpff_eigen INTERFACE)
    target_include_directories(rrtpff_eigen SYSTEM INTERFACE "${RRTPFF_EIGEN_SOURCE}")
    add_library(Eigen3::Eigen ALIAS rrtpff_eigen)
else()
    find_package(Eigen3 3.3 REQUIRED NO_MODULE)
endif()
if(RRTPFF_YAML_SOURCE)
    set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${RRTPFF_YAML_SOURCE}" "${CMAKE_BINARY_DIR}/yaml-cpp" EXCLUDE_FROM_ALL)
else()
    find_package(yaml-cpp REQUIRED)
endif()
add_library(rrtstar_df_pff STATIC pff/pff.cpp core/rrtstar.cpp simd/simd_backend.cpp simd/physical_batch.cpp simd/production_dispatch.cpp)
target_include_directories(rrtstar_df_pff PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(rrtstar_df_pff PUBLIC Eigen3::Eigen)
if(RRTSTARDFPFF_ENABLE_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64|i.86")
    target_sources(rrtstar_df_pff PRIVATE simd/simd_avx2.cpp simd/physical_avx2.cpp simd/production_avx2.cpp)
    target_compile_definitions(rrtstar_df_pff PRIVATE RRTPFF_HAS_AVX2=1)
    if(MSVC)
        set_source_files_properties(simd/simd_avx2.cpp simd/physical_avx2.cpp simd/production_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2;/fp:strict")
    else()
        set_source_files_properties(simd/simd_avx2.cpp simd/physical_avx2.cpp simd/production_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-ffp-contract=off")
    endif()
endif()
# Template call sites in benchmark/tests must share strict scalar arithmetic.
# AVX flags are NEVER exported to these translation units.
if(MSVC)
    target_compile_options(rrtstar_df_pff PUBLIC /fp:strict)
    target_compile_options(rrtstar_df_pff PRIVATE /W4 /permissive-)
else()
    target_compile_options(rrtstar_df_pff PUBLIC -ffp-contract=off)
    target_compile_options(rrtstar_df_pff PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(benchmark_pff benchmark/benchmark_pff.cpp)
target_link_libraries(benchmark_pff PRIVATE rrtstar_df_pff)
add_executable(benchmark benchmark/benchmark.cpp)
target_link_libraries(benchmark PRIVATE rrtstar_df_pff yaml-cpp)
# Data-only YAML adapter. DynoBench and FCL are NOT linked or invoked.
add_library(rrtpff_paper_data STATIC dynobench/data_adapter.cpp)
target_link_libraries(rrtpff_paper_data PUBLIC rrtstar_df_pff yaml-cpp)
target_sources(benchmark PRIVATE benchmark/robot_runner.cpp benchmark/production_runner.cpp)
target_compile_definitions(benchmark PRIVATE RRTPFF_WITH_PAPER_SIMD=1)
target_link_libraries(benchmark PRIVATE rrtpff_paper_data)
include(CTest)
if(BUILD_TESTING)
    foreach(test_name pff tree rewire_metadata end_to_end df dynobench simd search_improvements paper_fast physical_simd sphere_geometry paper_r2 float32 float_physical termination)
        add_executable(test_${test_name} tests/test_${test_name}.cpp)
        target_link_libraries(test_${test_name} PRIVATE rrtstar_df_pff rrtpff_paper_data)
        add_test(NAME ${test_name} COMMAND test_${test_name})
        set_tests_properties(${test_name} PROPERTIES TIMEOUT 60)
    endforeach()
    #add_executable(test_paper_profile tests/test_paper_profile.cpp)
    #target_link_libraries(test_paper_profile PRIVATE rrtpff_paper_data)
    #add_test(NAME paper_profile COMMAND test_paper_profile "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
